    // 🤖 自动从 API 拉取模型列表
    async function fetchRemoteModels() {
        const baseUrl = document.getElementById('api-input-url').value.trim();
        const apiKey = document.getElementById('api-input-key').value.trim();

        if(!apiKey) { showToast('请先填入 API Key'); return; }

        showToast('正在请求拉取模型列表中...');
        const cleanUrl = baseUrl.replace(/\/+$/, '');

        try {
            const res = await fetch(`${cleanUrl}/models`, {
                headers: { 'Authorization': `Bearer ${apiKey}` }
            });
            const data = await res.json();

            if(data && data.data && Array.isArray(data.data)) {
                const modelSelect = document.getElementById('api-fetched-models-select');
                modelSelect.innerHTML = '<option value="">-- 请选择拉取的模型 --</option>';

                data.data.forEach(m => {
                    modelSelect.innerHTML += `<option value="${m.id}">${m.id}</option>`;
                });

                modelSelect.style.display = 'block';
                showToast(`成功获取到 ${data.data.length} 个可用模型！`);
            } else {
                showToast('拉取失败，请检查 Base URL 或 Key');
            }
        } catch(e) {
            showToast('请求失败：' + e.message);
        }
    }

    async function runAiDiagnosis() {
        const activeChar = chars.find(c => c.id === activeCharId) || chars[0];
        const activeMask = masks.find(m => m.id === activeMaskId) || masks[0];

        if (!activeChar || !activeMask) {
            showToast('请先在"我的"和右上角分别创建并选择一个 User 面具与 AI Char 角色');
            return;
        }

        document.getElementById('ai-modal-title').textContent = `${activeChar.icon} ${activeChar.name} · 诊断分析`;
        const box = document.getElementById('ai-response-box');
        document.getElementById('modal-ai').classList.add('active');

        if(!apiConfig.apiKey) {
            box.innerHTML = `
                <div style="color:#e54d4d; margin-bottom:8px;">⚠️ 未检测到 API Key</div>
                <p>请在「我的」->「API 密钥设置」中填入 API Key。</p>
                <br>
                <p><strong>【${activeChar.name}】模拟诊断：</strong></p>
                <p>"${activeMask.name}，本月开销整体控制在合理范围内，请继续保持良好的记账习惯。"</p>
            `;
            return;
        }

        box.innerHTML = `✨ 正在连接 ${apiConfig.model}，${activeChar.name} 思考中...`;

        let totalOut = transactions.filter(t => t.type==='out').reduce((s,t) => s+t.amount, 0);
        let totalIn = transactions.filter(t => t.type==='in').reduce((s,t) => s+t.amount, 0);

        const prompt = `你现在角色扮演名为「${activeChar.name}」的 AI 伴侣/管家（性格语气：${activeChar.trait}）。
你的对话对象（用户）当前的身份面具是：「${activeMask.name}」（特点：${activeMask.trait}）。
用户本月财务概况：月度预算上限为 ${monthlyBudget} 元，当前实际支出 ${totalOut} 元，总收入 ${totalIn} 元。
请完全以「${activeChar.name}」的人设语气，对「${activeMask.name}」本月的消费情况做出简短、精致、有温度的针对性评价与理财建议（150字以内）。`;

        try {
            const res = await fetch(`${apiConfig.baseUrl.replace(/\/+$/, '')}/chat/completions`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Authorization': `Bearer ${apiConfig.apiKey}`
                },
                body: JSON.stringify({
                    model: apiConfig.model,
                    messages: [
                        { role: 'system', content: `你是 ${activeChar.name}。` },
                        { role: 'user', content: prompt }
                    ],
                    temperature: 0.7
                })
            });

            const data = await res.json();
            if(data.choices && data.choices[0]) {
                box.innerHTML = data.choices[0].message.content.replace(/\n/g, '<br>');
            } else {
                box.textContent = '接口响应异常：' + JSON.stringify(data);
            }
        } catch(err) {
            box.innerHTML = `<span style="color:#e54d4d;">请求失败：${err.message}。请检查网络或 CORS 跨域设置。</span>`;
        }
    }

    function calculateAssetDiff(assetName) {
        let diff = 0;
        transactions.forEach(t => {
            if (t.asset === assetName) {
                if (t.type === 'in') diff += t.amount;
                if (t.type === 'out') diff -= t.amount;
            }
        });
        return diff;
    }

    // ===== 📷 真实小票扫描：拍照/相册选图 + OCR 文字识别 =====
    function handleScanFile(event) {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function(e) {
            document.getElementById('scan-preview-box').innerHTML = `<img src="${e.target.result}" style="width:100%;height:100%;object-fit:cover;">`;
            runReceiptOCR(e.target.result);
        };
        reader.readAsDataURL(file);
    }

    async function runReceiptOCR(dataUrl) {
        const statusEl = document.getElementById('scan-status-text');
        const confirmBtn = document.getElementById('scan-confirm-btn');
        confirmBtn.style.display = 'none';

        if (typeof Tesseract === 'undefined') {
            statusEl.textContent = '⚠️ OCR 引擎加载失败（可能无网络），请稍后重试，或使用下方模拟识别。';
            return;
        }

        statusEl.textContent = '🔍 正在识别小票文字，请稍候...';

        try {
            const result = await Tesseract.recognize(dataUrl, 'chi_sim+eng', {
                logger: m => {
                    if (m.status === 'recognizing text' && typeof m.progress === 'number') {
                        statusEl.textContent = `🔍 正在识别中... ${Math.round(m.progress * 100)}%`;
                    }
                }
            });
            const text = result.data && result.data.text ? result.data.text : '';
            scanExtracted = parseReceiptText(text);

            if (scanExtracted.amount) {
                statusEl.innerHTML = `✅ 识别完成！金额：<strong>¥${scanExtracted.amount}</strong>${scanExtracted.memo ? '　备注：' + scanExtracted.memo : ''}<br><span style="font-size:11px;">识别结果仅供参考，下一步可修改后再保存</span>`;
            } else {
                statusEl.textContent = '未能自动识别出金额，点击"确认填入账目"后可手动补充，或重新拍摄更清晰的小票。';
            }
            confirmBtn.style.display = 'block';
        } catch (err) {
            statusEl.textContent = '识别失败：' + err.message + '。请检查网络，或使用下方模拟识别。';
        }
    }

    function parseReceiptText(text) {
        const lines = text.split(/\n/).map(l => l.trim()).filter(Boolean);
        let amount = null;

        // 优先寻找“合计/总计/实付/应付/金额”等关键词后面跟着的数字
        const keywordRegex = /(合计|总计|总额|实付|应付|消费金额|金额)[：:￥¥\s]*([0-9]+(\.[0-9]{1,2})?)/;
        for (const line of lines) {
            const m = line.match(keywordRegex);
            if (m) { amount = parseFloat(m[2]); break; }
        }

        // 没找到关键词时，退而求其次：取文本中所有形如 12.34 的数字里最大的一个
        if (amount === null) {
            const nums = [...text.matchAll(/[¥￥]?\s?([0-9]{1,6}\.[0-9]{2})\b/g)].map(m => parseFloat(m[1]));
            if (nums.length) amount = Math.max(...nums);
        }

        const memo = lines.length ? lines[0].slice(0, 20) : '';
        return { amount, memo };
    }

    function importData() {
        const jsonStr = prompt('请粘贴 JSON 备份数据：');
        if(jsonStr) {
            try {
                const p = JSON.parse(jsonStr);
                if(p.transactions) transactions = p.transactions;
                if(p.masks) masks = p.masks;
                if(p.chars) chars = p.chars;
                if(p.apiConfig) apiConfig = p.apiConfig;

                showToast('数据已恢复！');
            } catch(e) {
                alert('JSON 格式解析失败');
            }
        }
    }
